// 显示封装：源域27MHz、像素25.2MHz、DDR串行126MHz；锁定后各域分别释放复位。
// sample_tick是源域时钟使能，pcm为16位二补码；pixel域只读取整帧状态与双缓冲波形。
// UI_ENABLE默认0保留242bit旧接口，=1增加高10bit页/游标/曲目/控制键。
// COMPACT_UI默认0兼容旧图；=1隐含开启四页元数据并增加37位，共289位整包跨域。
// strum_mode[1:0]：0=KEY、1=STRUM、2=HYBRID；模式与其余状态整包跨域。
// HYBRID_MODE_ENABLE=1仅在compact分支最高位追加混合模式标志，不移动任何旧字段。
// knob_values从低到高每8位对应K1..K4；与页、模式、旧状态同次握手，只在帧边界发布。
// 低242bit打包与解包顺序必须匹配，见下方布局；不要为多位状态逐位双触发器同步。
// source_status由req/ack整包握手捕获，display_status仅frame_tick帧边界切换。
// 旧分支波形RAM一拍、字体一拍、RGB一拍，共三拍；COMPACT增加字符ROM一拍，共四拍。
// 两分支各自的字色、背景、RGB与DE/HS/VS共同延迟；不修改原波形地址/采样率。
// 复位立即断言且每域同步释放；PLL失锁清空显示状态但不复位音频层。
// peak保留接口但当前波形缩放取capture_peak；不要为了消除未用端口而改所有位置调用。
// 输出为兼容DVI的视频TMDS，没有HDMI音频包、HDCP或USB触控。

// Instrument status crosses domains as a coherent frame-boundary snapshot.
// The scope uses immutable RAM banks with its own request/ack handshake.
module hdmi_live_display #(parameter UI_ENABLE=0,parameter COMPACT_UI=0,
 parameter PIANO_ROLL_ENABLE=0,parameter HYBRID_MODE_ENABLE=0,
 parameter EXPRESSION_UI_ENABLE=0,parameter VISUALIZATION_ENABLE=0,
 parameter CUSTOM_SCORE_ENABLE=0)(

 input signed [8:0] pitch,input [5:0] voices,input [7:0] pressure0,pressure1,flags,

 // 测试页新增 78 位：1+6+32+32+1+5+1；全部由音频源时钟域的父模块提供。
 // test_active 表示测试引擎持有音频通路，包含进入/退出交接，不能仅接测试按键。
 // requested 为请求振荡器数（0..32）；busy/gate 的 bit i 对应第 i 路（0..31）。
 // busy 包含释放尾音；gate 是当前门控请求，两者必须分别显示。
 // sweep/index 为扫频模式和当前 0..31 索引；overrun 是测试调度超时状态。
 // voices 在测试期间应由父模块对 busy 计数，保持“请求数/实际活动数”的区别。

 input [4:0] test_index,input test_overrun,
 input [1:0] ui_page,ui_cursor,ui_song,input [3:0] ui_controls,

 // ADC低16位=FSR0/高16位=FSR1，保留ADS二补码原始码，用HEX显示而非kg。
 // velocity是最近有效NoteOn的原始MIDI力度0..127；bend是-256..255控制值，

 // V036：最终PCM的FFT与独立声部0/1的XY，均为只读观察，独立双缓冲CDC。
 input signed [15:0] visual_voice0,visual_voice1,
 // Broadcast unchanged to source guide + pixel RAM. Config is SOURCE-domain;
 // renderer owns the held-bus/toggle transfer and atomic frame publication.
 // ACK remains a pixel-domain level: top/controller must synchronise TWICE
 // into clk27 and reject BEGIN while ack != custom_config[31].
 input custom_write_clk,input custom_write_en,
 input [8:0] custom_write_addr,input [31:0] custom_write_data,
 input [31:0] custom_config,output custom_bank_ack);
 localparam USE_COMPACT=COMPACT_UI || PIANO_ROLL_ENABLE;

 hdmi_clocks u_clocks(clk27,rst_n,serial_clk,pixel_clk,pll_locked);

 (* syn_preserve=1 *) reg [2:0] source_reset_sync,pixel_reset_sync;

  if(!common_reset_n)source_reset_sync<=0;
  else source_reset_sync<={source_reset_sync[1:0],1'b1};
 always @(posedge pixel_clk or negedge common_reset_n)
  if(!common_reset_n)pixel_reset_sync<=0;
  else pixel_reset_sync<={pixel_reset_sync[1:0],1'b1};

 // 原字段顺序不变；测试字段追加到低 78 位，打包和解包严格使用同一顺序。

 // [6] sweep，[5:1] index，[0] overrun；仅在帧边界使用已应答的整包数据。
 // COMPACT implies the four-page metadata even when UI_ENABLE is omitted.
 // 289-bit compact packet: [288:287] control page, [286:285] FX,
 // [284] strum, [283:252] K4..K1. Low252 bits remain exactly compatible.
 // Feature=0 preserves 242/252/289 bits exactly, with no extra CDC registers.
 // Feature=1 implies compact and appends49 bits to the SAME bundled handshake:
 // [337:306] piano_keys, [305] outside, [304:293] tick, [292:291] song,
 // [290] active, [289] done. Low289 bits remain the compact legacy layout.
 localparam BASE_STATUS_WIDTH=164+78+((UI_ENABLE || USE_COMPACT) ? 10 : 0)+(USE_COMPACT ? 37 : 0);
 // The old strum bit means optical input enabled (STRUM or HYBRID). The
 // optional new top bit distinguishes HYBRID in the same coherent snapshot.
 // Parameter0 preserves the historical widths242/252/289/338 exactly;
 // parameter1 adds one bit only for compact:290, or339 with piano enabled.

 // [396:365]ADC [364:363]cal [362:355]velocity [354:346]bend [345:339]mod。
 localparam USE_EXPRESSION=USE_COMPACT && EXPRESSION_UI_ENABLE;
 localparam EXPRESSION_STATUS_WIDTH=HYBRID_STATUS_WIDTH+(USE_EXPRESSION ? 58 : 0);
 localparam USE_CUSTOM=PIANO_ROLL_ENABLE && CUSTOM_SCORE_ENABLE;

 wire [PIANO_STATUS_WIDTH-1:0] piano_source_status;
 wire [HYBRID_STATUS_WIDTH-1:0] hybrid_source_status;
 wire [EXPRESSION_STATUS_WIDTH-1:0] expression_source_status;

 end else if(UI_ENABLE)begin:source_ui_packet
  assign base_source_status={ui_page,ui_cursor,ui_song,ui_controls,legacy_source_status};
 end else begin:source_legacy_packet
  assign base_source_status=legacy_source_status;
 end endgenerate
 generate if(PIANO_ROLL_ENABLE)begin:source_piano_packet
  assign piano_source_status={piano_keys,piano_outside,roll_tick,roll_song,roll_active,roll_done,base_source_status};
 end else begin:source_no_piano_packet
  assign piano_source_status=base_source_status;
 end endgenerate
 generate if(USE_HYBRID)begin:source_hybrid_packet
  assign hybrid_source_status={(strum_mode==2'd2),piano_source_status};
 end else begin:source_no_hybrid_packet
  assign hybrid_source_status=piano_source_status;
 end endgenerate
 generate if(USE_EXPRESSION)begin:source_expression_packet
  assign expression_source_status={expression_adc,expression_calibrated,midi_velocity,
   midi_bend,midi_modulation,hybrid_source_status};
 end else begin:source_no_expression_packet
  assign expression_source_status=hybrid_source_status;
 end endgenerate
 wire d_roll_custom_bank;
 generate if(USE_CUSTOM)begin:custom_timeline_tag
  reg source_bank;
  // Same edge as guide's synchronous config detection/restart. Thus a held
  // status snapshot sees either old tick+old tag or restarted tick+new tag.
  always @(posedge clk27 or negedge source_reset_sync[2])
   if(!source_reset_sync[2])source_bank<=0;
   else source_bank<=custom_config[31];
  assign source_status={source_bank,expression_source_status};
  assign d_roll_custom_bank=display_status[STATUS_WIDTH-1];
 end else begin:no_custom_timeline_tag
  assign source_status=expression_source_status;assign d_roll_custom_bank=0;
 end endgenerate
 wire [STATUS_WIDTH-1:0] display_status;
 hdmi_status_cdc #(.WIDTH(STATUS_WIDTH)) u_status(clk27,source_reset_sync[2],source_status,
  pixel_clk,video_rst_n,frame_tick,display_status);
 wire [11:0] d_keys,d_cue,d_next,d_wrong;
 wire [3:0] d_timbre;
 wire [6:0] d_note;
 wire d_note_valid,d_active,d_done,d_clip;
 wire [15:0] d_frequency;

 wire signed [7:0] d_level;
 wire signed [8:0] d_pitch;
 wire [5:0] d_voices,d_step,d_total;
 wire [1:0] d_beats;
 wire d_test_active,d_test_sweep,d_test_overrun;
 wire [5:0] d_test_requested;
 wire [31:0] d_test_busy,d_test_gate;
 wire [4:0] d_test_index;
 assign {d_keys,d_timbre,d_note,d_note_valid,d_frequency,d_volume,d_level,d_peak,
  d_pitch,d_voices,d_pressure0,d_pressure1,d_flags,d_active,d_done,d_cue,d_next,
  d_wrong,d_step,d_total,d_progress,d_beats,d_clip,
  d_test_active,d_test_requested,d_test_busy,d_test_gate,
  d_test_sweep,d_test_index,d_test_overrun}=display_status[241:0];
 // d_test_* 只在帧边界更新；过渡期间仍显示 OSC TEST，避免半帧混入普通页面。
 // Physical clipping and audio engine overrun share the ALERT flag in the UI.
 wire [7:0] alert_flags=d_flags | (d_clip?8'h80:8'h00);
 wire [7:0] scope_addr=(x-10'd32)>>1;
 wire signed [15:0] scope_current,scope_previous;
 wire [3:0] scope_gain;
 wire scope_valid;
 wave_scope u_scope(clk27,source_reset_sync[2],sample_tick,pcm,
  pixel_clk,video_rst_n,frame_tick,scope_addr,scope_current,scope_previous,scope_gain,scope_valid);

 reg [9:0] draw_x,draw_y;

 always @(posedge pixel_clk or negedge video_rst_n)begin
  if(!video_rst_n)begin draw_x<=0;draw_y<=0;draw_de<=0;draw_hs<=1;draw_vs<=1;end
  else begin draw_x<=x;draw_y<=y;draw_de<=de;draw_hs<=hsync;draw_vs<=vsync;end

 wire [7:0] r,g,b;
 wire [1:0] d_ui_page,d_ui_cursor,d_ui_song;

  .test_active(d_test_active),.test_requested(d_test_requested),
  .test_busy(d_test_busy),.test_gate(d_test_gate),.test_sweep(d_test_sweep),
  .test_index(d_test_index),.test_overrun(d_test_overrun),
  .pixel_clk(pixel_clk),.rst_n(video_rst_n),
  .ui_page(d_ui_page),.ui_cursor(d_ui_cursor),.ui_song(d_ui_song),.ui_controls(d_ui_controls));
 end endgenerate
 wire [1:0] d_control_page,d_fx_mode;
 wire d_strum_enabled;
 wire [1:0] d_strum_mode;
 wire [31:0] d_knob_values;
 wire [31:0] d_piano_keys;
